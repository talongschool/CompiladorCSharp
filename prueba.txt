precio unitario y aplicar un 20% de descuento si el cliente compra 10 o más unidades siendo 
miembro VIP, un 10% de descuento si compra 5 o más unidades en cualquier otro caso, o sin 
descuento si lleva menos de 5 unidades, probando la función en el main con 12 unidades a un

float calcularTotalCompra(int unidades, float precioUnitario, bool esMiembroVip) {
    float subtotal = unidades * precioUnitario;
    float descuento = 0.0;
    
    // comentariopqsibro
    if (unidades >= 10 && esMiembroVip == true) {
        descuento = subtotal * 0.20;
    } else {
        if (unidades >= 5) {
            descuento = subtotal * 0.10;
        } else {
            descuento = 0.0;
        }
    }

    float totalFinal = subtotal - descuento;
    return totalFinal;
}
